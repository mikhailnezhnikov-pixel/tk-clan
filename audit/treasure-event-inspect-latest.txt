def release_notes() -> str:

def release_manifest(current_version: str, download_token: str = "") -> dict:
    latest = release_version()

def rate_allowed(key: str, limit: int, window: int) -> bool:
    now = utc_now()
    with RATE_LOCK:
        recent = [stamp for stamp in RATE_BUCKETS.get(key, []) if stamp > now - window]
        if len(recent) >= limit:
            RATE_BUCKETS[key] = recent
            return False
        recent.append(now)
        RATE_BUCKETS[key] = recent
        if len(RATE_BUCKETS) > 10000:
            for old_key in list(RATE_BUCKETS)[:2000]:
                if not RATE_BUCKETS[old_key] or RATE_BUCKETS[old_key][-1] <= now - 3600:
                    RATE_BUCKETS.pop(old_key, None)
        return True

def license_check(player_id: str, device_id: str, version: str, client_ip: str = "") -> tuple[int, dict]:
    now = utc_now()
    with db_session() as db:
        license_row = db.execute("SELECT * FROM licenses WHERE player_id=?", (player_id,)).fetchone()
        if not license_row:
            return HTTPStatus.FORBIDDEN, {"allowed": False, "reason": "not_allowed", "player_id": player_id}
        if not license_row["active"]:
            return HTTPStatus.FORBIDDEN, {"allowed": False, "reason": "blocked", "player_id": player_id}
        if license_row["expires_at"] and license_row["expires_at"] < now:
            return HTTPStatus.FORBIDDEN, {"allowed": False, "reason": "expired", "player_id": player_id}
        device = db.execute("SELECT 1 FROM devices WHERE player_id=? AND device_id=?",
                            (player_id, device_id)).fetchone()
        if not device:
            db.execute("INSERT INTO devices(player_id,device_id,first_seen,last_seen,script_version) VALUES(?,?,?,?,?)",
                       (player_id, device_id, now, now, version[:32]))
        else:
            db.execute("UPDATE devices SET last_seen=?,script_version=? WHERE player_id=? AND device_id=?",
                       (now, version[:32], player_id, device_id))
        record_access(db, player_id, device_id, version, client_ip)
        token = sign_token(f"{player_id}:{device_id}", 900)
        update = release_manifest(version)
        if update["available"]:
            update = release_manifest(version, issue_update_token(player_id, device_id, db))
        if update["required"]:
            return HTTPStatus.UPGRADE_REQUIRED, {"allowed": False, "reason": "update_required",
                                                 "player_id": player_id, "minimum_version": MIN_SCRIPT_VERSION,
                                                 "update": update}
        return HTTPStatus.OK, {"allowed": True, "player_id": player_id, "token": token,
                               "check_after": 300, "expires_at": license_row["expires_at"], "update": update,
                               "public_collector_auth_sync": public_collector_auth_sync_allowed(player_id)}


def normalize_business_plans(value: object, allow_components: bool = True) -> list[dict]:

        raise ValueError("invalid business plans")
    result = []
    seen = set()
    for item in value:
        if not isinstance(item, dict):
            raise ValueError("invalid business plan")
        plan_id = str(item.get("plan_id", "")).strip()
        item_id = str(item.get("item_id", "")).strip()
        if not plan_id.startswith(("recipe_", "mf_craftlot_")) or not ID_RE.fullmatch(plan_id) or not ID_RE.fullmatch(item_id):
            raise ValueError("invalid business plan")
        if plan_id in seen:
            continue
        seen.add(plan_id)
        tier = item.get("tier")
        rarity = item.get("rarity")
        quality = item.get("quality")
        tier = int(tier) if tier is not None else None
        rarity = int(rarity) if rarity is not None else None
        quality = int(quality) if quality is not None else None
        if tier is not None and not 1 <= tier <= 99:
            raise ValueError("invalid tier")
        if rarity is not None and not 0 <= rarity <= 99:
            raise ValueError("invalid rarity")
        if quality is not None and not 0 <= quality <= 999:
            raise ValueError("invalid quality")
        bonuses = item.get("bonuses") if isinstance(item.get("bonuses"), list) else []
        bonuses = [{"id":str(row.get("id", ""))[:160], "caption":str(row.get("caption", ""))[:300],
                    "value":str(row.get("value", ""))[:80], "icon":str(row.get("icon", ""))[:600]}
                   for row in bonuses[:12] if isinstance(row, dict)]
        components = item.get("components") if allow_components and isinstance(item.get("components"), list) else []
        components = normalize_business_plans(components[:8], False) if components else []
        result.append({"plan_id": plan_id, "item_id": item_id, "tier": tier, "rarity": rarity, "quality": quality,
                       "name":str(item.get("name", ""))[:300], "description":str(item.get("description", ""))[:1000],
                       "icon":str(item.get("icon", ""))[:600], "color":str(item.get("color", ""))[:32],

    return result
