def utc_now() -> int:

def iso_time(timestamp: int | None) -> str:
    if not timestamp:
        return "—"
    return datetime.fromtimestamp(timestamp, timezone.utc).strftime("%d.%m.%Y %H:%M UTC")


def active_rumor_route_date(now: datetime | None = None) -> str:
    """Return the route date for the 21:00-20:59 game-day window."""
    moment = now or datetime.now(timezone.utc)
    if moment.tzinfo is None:
        moment = moment.replace(tzinfo=timezone.utc)
    local = moment.astimezone(RUMOR_TIMEZONE)
    if local.hour < 21:
        local -= timedelta(days=1)
    return local.strftime("%Y-%m-%d")


def version_tuple(value: str) -> tuple[int, ...] | None:
    """Parse a numeric dotted version and ignore harmless suffixes."""
    match = re.match(r"^\s*(\d+(?:\.\d+)*)", value or "")
    if not match:
        return None
    return tuple(int(part) for part in match.group(1).split("."))


def version_at_least(value: str, minimum: str) -> bool:
    current = version_tuple(value)
    required = version_tuple(minimum)
    if current is None or required is None:
        return False
    width = max(len(current), len(required))
    return current + (0,) * (width - len(current)) >= required + (0,) * (width - len(required))


def release_version() -> str:
    try:
        with open(RELEASE_PATH, "r", encoding="utf-8-sig") as source:
            head = source.read(4096)
        match = re.search(r"^//\s*@version\s+([^\s]+)", head, re.MULTILINE)
        return match.group(1).strip() if match else MIN_SCRIPT_VERSION
    except OSError:
        return MIN_SCRIPT_VERSION


def release_sha256() -> str:
    try:
        digest = hashlib.sha256()
        with open(RELEASE_PATH, "rb") as source:
            for chunk in iter(lambda: source.read(131072), b""):
                digest.update(chunk)
        return digest.hexdigest()
    except OSError:
        return ""


def release_notes() -> str:
    """Read version-specific notes embedded in the published userscript."""
    try:
        with open(RELEASE_PATH, "r", encoding="utf-8-sig") as source:
            head = source.read(16384)
        notes = [match.strip() for match in re.findall(
            r"^//\s*@release-note\s+(.+?)\s*$", head, re.MULTILINE
        ) if match.strip()]
        if notes:
            return "\n".join(f"• {note}" for note in notes)[:2000]
    except OSError:
        pass
    return RELEASE_NOTES


def release_filename() -> str:
    version = re.sub(r"[^0-9A-Za-z._-]", "_", release_version())
    return f"HamsterKingMobile_v{version}.user.js"


def release_manifest(current_version: str, download_token: str = "") -> dict:
    latest = release_version()
    available = not version_at_least(current_version, latest)
    required = not version_at_least(current_version, MIN_SCRIPT_VERSION)
    result = {"current_version": current_version, "latest_version": latest,
              "minimum_version": MIN_SCRIPT_VERSION, "available": available,
              "required": required, "notes": release_notes() if available else "",
              "sha256": release_sha256() if available else ""}
    if download_token and available:
        result["download_url"] = f"{PUBLIC_BASE}/updates/{release_filename()}?token={download_token}"
    return result


def rate_allowed(key: str, limit: int, window: int) -> bool:
    now = utc_now()
    with RATE_LOCK:
        recent = [stamp for stamp in RATE_BUCKETS.get(key, []) if stamp > now - window]
        if len(recent) >= limit:
            RATE_BUCKETS[key] = recent
            return False
        recent.append(now)
        RATE_BUCKETS[key] = recent
        if len(RATE_BUCKETS) > 10000:
            for old_key in list(RATE_BUCKETS)[:2000]:
                if not RATE_BUCKETS[old_key] or RATE_BUCKETS[old_key][-1] <= now - 3600:
                    RATE_BUCKETS.pop(old_key, None)
        return True


def contributor_id(player_id: str) -> str:
    if player_id in {"historical-har", "kokkaras-import"} or player_id.startswith("source-"):
        return player_id
    digest = hmac.new(SIGNING_SECRET.encode(), f"contributor:{player_id}".encode(), hashlib.sha256).hexdigest()
    return f"anon-{digest[:24]}"


def connect() -> sqlite3.Connection:
    db = sqlite3.connect(DB_PATH, timeout=10)
    db.row_factory = sqlite3.Row
    db.execute("PRAGMA journal_mode=WAL")
    db.execute("PRAGMA foreign_keys=ON")
    return db


@contextmanager
def db_session():
    db = connect()
    try:
        yield db
        db.commit()
    finally:
        db.close()

PIT_TYPES = {"normal", "boss", "gang"}


def normalize_pit_observations(value: object) -> list[dict]:
    if not isinstance(value, list) or len(value) > 80:
        raise ValueError("invalid pit observations")
    result: list[dict] = []
    for raw in value:
        if not isinstance(raw, dict):
            continue
        pit_type = str(raw.get("pit_type", "")).strip().lower()
        try:
            level = int(raw.get("level"))
            enemy_power = int(raw.get("enemy_power"))
        except (TypeError, ValueError):
            continue
        if pit_type not in PIT_TYPES or not 1 <= level <= 10000 or not 1 <= enemy_power <= 10**18:
            continue
        player_power = raw.get("player_power")
        try:
            player_power = int(player_power) if player_power not in (None, "") else None
        except (TypeError, ValueError):
            player_power = None
        if player_power is not None and not 0 <= player_power <= 10**18:
            player_power = None
        winrate = raw.get("winrate")
        try:
            winrate = float(winrate) if winrate not in (None, "") else None
        except (TypeError, ValueError):
            winrate = None
        if winrate is not None:
            if 1 < winrate <= 100:
                winrate /= 100
            if not 0 <= winrate <= 1:
                winrate = None
        result.append({"pit_type": pit_type, "level": level, "enemy_power": enemy_power,
                       "player_power": player_power, "winrate": winrate})
    return result

def version_tuple(value: str) -> tuple[int, ...] | None:
    """Parse a numeric dotted version and ignore harmless suffixes."""
    match = re.match(r"^\s*(\d+(?:\.\d+)*)", value or "")
    if not match:
        return None
    return tuple(int(part) for part in match.group(1).split("."))


def version_at_least(value: str, minimum: str) -> bool:
    current = version_tuple(value)
    required = version_tuple(minimum)
    if current is None or required is None:
        return False
    width = max(len(current), len(required))
    return current + (0,) * (width - len(current)) >= required + (0,) * (width - len(required))

def rate_allowed(key: str, limit: int, window: int) -> bool:
    now = utc_now()
    with RATE_LOCK:
        recent = [stamp for stamp in RATE_BUCKETS.get(key, []) if stamp > now - window]
        if len(recent) >= limit:
            RATE_BUCKETS[key] = recent
            return False
        recent.append(now)
        RATE_BUCKETS[key] = recent
        if len(RATE_BUCKETS) > 10000:
            for old_key in list(RATE_BUCKETS)[:2000]:
                if not RATE_BUCKETS[old_key] or RATE_BUCKETS[old_key][-1] <= now - 3600:
                    RATE_BUCKETS.pop(old_key, None)
        return True

PIT_TYPES = {"normal", "boss", "gang"}


def normalize_pit_observations(value: object) -> list[dict]:
    if not isinstance(value, list) or len(value) > 80:
        raise ValueError("invalid pit observations")
    result: list[dict] = []
    for raw in value:
        if not isinstance(raw, dict):
            continue
        pit_type = str(raw.get("pit_type", "")).strip().lower()
        try:
            level = int(raw.get("level"))
            enemy_power = int(raw.get("enemy_power"))
        except (TypeError, ValueError):
            continue
        if pit_type not in PIT_TYPES or not 1 <= level <= 10000 or not 1 <= enemy_power <= 10**18:
            continue
        player_power = raw.get("player_power")
        try:
            player_power = int(player_power) if player_power not in (None, "") else None
        except (TypeError, ValueError):
            player_power = None
        if player_power is not None and not 0 <= player_power <= 10**18:
            player_power = None
        winrate = raw.get("winrate")
        try:
            winrate = float(winrate) if winrate not in (None, "") else None
        except (TypeError, ValueError):
            winrate = None
        if winrate is not None:
            if 1 < winrate <= 100:
                winrate /= 100
            if not 0 <= winrate <= 1:
                winrate = None
        result.append({"pit_type": pit_type, "level": level, "enemy_power": enemy_power,
                       "player_power": player_power, "winrate": winrate})
    return result


def submit_pit_observations(player_id: str, value: object) -> dict:
    observations = normalize_pit_observations(value)
    contributor = contributor_id(player_id)
    now = utc_now()
    with db_session() as db:
        for row in observations:
            db.execute("""INSERT INTO pit_power_observations(
                            pit_type,level,enemy_power,player_power,winrate,contributor_id,samples,first_seen,last_seen)
                          VALUES(?,?,?,?,?,?,1,?,?)
                          ON CONFLICT(pit_type,level,enemy_power,contributor_id) DO UPDATE SET
                            player_power=COALESCE(excluded.player_power,pit_power_observations.player_power),
                            winrate=COALESCE(excluded.winrate,pit_power_observations.winrate),
                            samples=pit_power_observations.samples+1,last_seen=excluded.last_seen""",
                       (row["pit_type"], row["level"], row["enemy_power"], row["player_power"],
                        row["winrate"], contributor, now, now))
        total = db.execute("SELECT COUNT(*) FROM pit_power_observations").fetchone()[0]
    return {"ok": True, "accepted": len(observations), "total": total}


def pit_power_catalog() -> list[dict]:
    with db_session() as db:
        rows = db.execute("""SELECT pit_type,level,enemy_power,
                              COUNT(DISTINCT contributor_id) AS players,SUM(samples) AS samples,
                              AVG(winrate) AS observed_winrate,MAX(last_seen) AS last_seen
                           FROM pit_power_observations
                           GROUP BY pit_type,level,enemy_power
                           ORDER BY level,pit_type,players DESC,samples DESC""").fetchall()
    return [dict(row) for row in rows]

        if len(RATE_BUCKETS) > 10000:
            for old_key in list(RATE_BUCKETS)[:2000]:
                if not RATE_BUCKETS[old_key] or RATE_BUCKETS[old_key][-1] <= now - 3600:
                    RATE_BUCKETS.pop(old_key, None)
        return True


def contributor_id(player_id: str) -> str:
    if player_id in {"historical-har", "kokkaras-import"} or player_id.startswith("source-"):
        return player_id
    digest = hmac.new(SIGNING_SECRET.encode(), f"contributor:{player_id}".encode(), hashlib.sha256).hexdigest()
    return f"anon-{digest[:24]}"


def connect() -> sqlite3.Connection:
    db = sqlite3.connect(DB_PATH, timeout=10)
    db.row_factory = sqlite3.Row
    db.execute("PRAGMA journal_mode=WAL")
    db.execute("PRAGMA foreign_keys=ON")
    return db


@contextmanager
def db_session():
    db = connect()
    try:
        yield db
        db.commit()
    finally:
        db.close()

PIT_TYPES = {"normal", "boss", "gang"}


def normalize_pit_observations(value: object) -> list[dict]:
    if not isinstance(value, list) or len(value) > 80:
        raise ValueError("invalid pit observations")
    result: list[dict] = []
    for raw in value:
        if not isinstance(raw, dict):
            continue
        pit_type = str(raw.get("pit_type", "")).strip().lower()
        try:
            level = int(raw.get("level"))
            enemy_power = int(raw.get("enemy_power"))
        except (TypeError, ValueError):
            continue
        if pit_type not in PIT_TYPES or not 1 <= level <= 10000 or not 1 <= enemy_power <= 10**18:
            continue
        player_power = raw.get("player_power")
        try:
            player_power = int(player_power) if player_power not in (None, "") else None
        except (TypeError, ValueError):
            player_power = None
        if player_power is not None and not 0 <= player_power <= 10**18:
            player_power = None
        winrate = raw.get("winrate")
        try:
            winrate = float(winrate) if winrate not in (None, "") else None
        except (TypeError, ValueError):
            winrate = None
        if winrate is not None:
            if 1 < winrate <= 100:
                winrate /= 100
            if not 0 <= winrate <= 1:
                winrate = None
        result.append({"pit_type": pit_type, "level": level, "enemy_power": enemy_power,
                       "player_power": player_power, "winrate": winrate})
    return result


def submit_pit_observations(player_id: str, value: object) -> dict:
    observations = normalize_pit_observations(value)
    contributor = contributor_id(player_id)

                          VALUES(?,?,?,?,?,?,1,?,?)
                          ON CONFLICT(pit_type,level,enemy_power,contributor_id) DO UPDATE SET
                            player_power=COALESCE(excluded.player_power,pit_power_observations.player_power),
                            winrate=COALESCE(excluded.winrate,pit_power_observations.winrate),
                            samples=pit_power_observations.samples+1,last_seen=excluded.last_seen""",
                       (row["pit_type"], row["level"], row["enemy_power"], row["player_power"],
                        row["winrate"], contributor, now, now))

def pit_power_catalog() -> list[dict]:
    with db_session() as db:
        rows = db.execute("""SELECT pit_type,level,enemy_power,
                              COUNT(DISTINCT contributor_id) AS players,SUM(samples) AS samples,
                              AVG(winrate) AS observed_winrate,MAX(last_seen) AS last_seen
                           FROM pit_power_observations
                           GROUP BY pit_type,level,enemy_power
                           ORDER BY level,pit_type,players DESC,samples DESC""").fetchall()
    return [dict(row) for row in rows]


def normalize_clan_skill_query(clan_key: object, week_key: object) -> tuple[str, str]:
    clan = str(clan_key or "").strip()
    week = str(week_key or "").strip()
    if not ID_RE.fullmatch(clan):
        raise ValueError("invalid clan key")
    try:
        parsed_week = datetime.strptime(week, "%Y-%m-%d")
    except (TypeError, ValueError) as error:
        raise ValueError("invalid week key") from error
    if parsed_week.weekday() != 0:
        raise ValueError("week key must be Monday")
    return clan, week


def normalize_clan_skill_submission(clan_key: object, week_key: object,
                                    nickname: object, levels: object) -> tuple[str, str, str, dict[str, int]]:
    clan, week = normalize_clan_skill_query(clan_key, week_key)
    name = str(nickname or "").strip()[:100]
    if not name:
        raise ValueError("nickname is required")
    if not isinstance(levels, dict) or len(levels) > 300:
        raise ValueError("invalid skill levels")
    normalized: dict[str, int] = {}
    for raw_id, raw_value in levels.items():
        line_id = str(raw_id or "").strip()
        if not ID_RE.fullmatch(line_id):
            continue
        try:
            value = int(raw_value)
        except (TypeError, ValueError):
            continue
        if 0 <= value <= 100_000:
            normalized[line_id] = value
    if not normalized:
        raise ValueError("skill levels are empty")
    return clan, week, name, normalized


def clan_skill_players(clan_key: str, week_key: str) -> list[dict]:
    clan, week = normalize_clan_skill_query(clan_key, week_key)
    with db_session() as db:
        rows = db.execute("""SELECT nickname,levels_json,scanned_at
                             FROM clan_skill_snapshots
                             WHERE clan_key=? AND week_key=?""", (clan, week)).fetchall()
    players = []
    for row in rows:
        try:
            levels = json.loads(row["levels_json"])
        except (json.JSONDecodeError, TypeError):
            levels = {}
        levels = {str(key): max(0, int(value)) for key, value in levels.items()
                  if isinstance(value, (int, float)) and not isinstance(value, bool)}
        players.append({"nickname": row["nickname"], "levels": levels,
                        "total": sum(levels.values()), "scanned_at": row["scanned_at"]})
    players.sort(key=lambda row: (-row["total"], row["nickname"].casefold()))
    return players

def utc_now() -> int:

def iso_time(timestamp: int | None) -> str:
    if not timestamp:
        return "—"
    return datetime.fromtimestamp(timestamp, timezone.utc).strftime("%d.%m.%Y %H:%M UTC")


def active_rumor_route_date(now: datetime | None = None) -> str:
    """Return the route date for the 21:00-20:59 game-day window."""
    moment = now or datetime.now(timezone.utc)
    if moment.tzinfo is None:
        moment = moment.replace(tzinfo=timezone.utc)
    local = moment.astimezone(RUMOR_TIMEZONE)
    if local.hour < 21:
        local -= timedelta(days=1)
    return local.strftime("%Y-%m-%d")


def version_tuple(value: str) -> tuple[int, ...] | None:
    """Parse a numeric dotted version and ignore harmless suffixes."""
    match = re.match(r"^\s*(\d+(?:\.\d+)*)", value or "")
    if not match:
        return None
    return tuple(int(part) for part in match.group(1).split("."))


def version_at_least(value: str, minimum: str) -> bool:
    current = version_tuple(value)
    required = version_tuple(minimum)
    if current is None or required is None:
        return False
    width = max(len(current), len(required))
    return current + (0,) * (width - len(current)) >= required + (0,) * (width - len(required))


def release_version() -> str:
    try:
        with open(RELEASE_PATH, "r", encoding="utf-8-sig") as source:
            head = source.read(4096)
        match = re.search(r"^//\s*@version\s+([^\s]+)", head, re.MULTILINE)
        return match.group(1).strip() if match else MIN_SCRIPT_VERSION
    except OSError:
        return MIN_SCRIPT_VERSION


def release_sha256() -> str:
    try:
        digest = hashlib.sha256()
        with open(RELEASE_PATH, "rb") as source:
            for chunk in iter(lambda: source.read(131072), b""):
                digest.update(chunk)
        return digest.hexdigest()
    except OSError:
        return ""


def release_notes() -> str:
    """Read version-specific notes embedded in the published userscript."""
    try:
        with open(RELEASE_PATH, "r", encoding="utf-8-sig") as source:
            head = source.read(16384)
        notes = [match.strip() for match in re.findall(
            r"^//\s*@release-note\s+(.+?)\s*$", head, re.MULTILINE
        ) if match.strip()]
        if notes:
            return "\n".join(f"• {note}" for note in notes)[:2000]
    except OSError:
        pass
    return RELEASE_NOTES


def release_filename() -> str:
    version = re.sub(r"[^0-9A-Za-z._-]", "_", release_version())
    return f"HamsterKingMobile_v{version}.user.js"


def release_manifest(current_version: str, download_token: str = "") -> dict:
    latest = release_version()
    available = not version_at_least(current_version, latest)
    required = not version_at_least(current_version, MIN_SCRIPT_VERSION)
    result = {"current_version": current_version, "latest_version": latest,
              "minimum_version": MIN_SCRIPT_VERSION, "available": available,
              "required": required, "notes": release_notes() if available else "",
              "sha256": release_sha256() if available else ""}
    if download_token and available:
        result["download_url"] = f"{PUBLIC_BASE}/updates/{release_filename()}?token={download_token}"
    return result


def rate_allowed(key: str, limit: int, window: int) -> bool:
    now = utc_now()
    with RATE_LOCK:
        recent = [stamp for stamp in RATE_BUCKETS.get(key, []) if stamp > now - window]
        if len(recent) >= limit:
            RATE_BUCKETS[key] = recent
            return False
        recent.append(now)
        RATE_BUCKETS[key] = recent
        if len(RATE_BUCKETS) > 10000:
            for old_key in list(RATE_BUCKETS)[:2000]:
                if not RATE_BUCKETS[old_key] or RATE_BUCKETS[old_key][-1] <= now - 3600:
                    RATE_BUCKETS.pop(old_key, None)
        return True


def contributor_id(player_id: str) -> str:
    if player_id in {"historical-har", "kokkaras-import"} or player_id.startswith("source-"):
        return player_id
    digest = hmac.new(SIGNING_SECRET.encode(), f"contributor:{player_id}".encode(), hashlib.sha256).hexdigest()
    return f"anon-{digest[:24]}"


def connect() -> sqlite3.Connection:
    db = sqlite3.connect(DB_PATH, timeout=10)
    db.row_factory = sqlite3.Row
    db.execute("PRAGMA journal_mode=WAL")
    db.execute("PRAGMA foreign_keys=ON")
    return db


@contextmanager
def db_session():
    db = connect()
    try:
        yield db
        db.commit()
    finally:
        db.close()

PIT_TYPES = {"normal", "boss", "gang"}


def normalize_pit_observations(value: object) -> list[dict]:
    if not isinstance(value, list) or len(value) > 80:
        raise ValueError("invalid pit observations")
    result: list[dict] = []
    for raw in value:
        if not isinstance(raw, dict):
            continue
        pit_type = str(raw.get("pit_type", "")).strip().lower()
        try:
            level = int(raw.get("level"))
            enemy_power = int(raw.get("enemy_power"))
        except (TypeError, ValueError):
            continue
        if pit_type not in PIT_TYPES or not 1 <= level <= 10000 or not 1 <= enemy_power <= 10**18:
            continue
        player_power = raw.get("player_power")
        try:
            player_power = int(player_power) if player_power not in (None, "") else None
        except (TypeError, ValueError):
            player_power = None
        if player_power is not None and not 0 <= player_power <= 10**18:
            player_power = None
        winrate = raw.get("winrate")
        try:
            winrate = float(winrate) if winrate not in (None, "") else None
        except (TypeError, ValueError):
            winrate = None
        if winrate is not None:
            if 1 < winrate <= 100:
                winrate /= 100
            if not 0 <= winrate <= 1:
                winrate = None
        result.append({"pit_type": pit_type, "level": level, "enemy_power": enemy_power,
                       "player_power": player_power, "winrate": winrate})
    return result


def submit_pit_observations(player_id: str, value: object) -> dict:
    observations = normalize_pit_observations(value)
    contributor = contributor_id(player_id)
    now = utc_now()
    with db_session() as db:
        for row in observations:
            db.execute("""INSERT INTO pit_power_observations(
                            pit_type,level,enemy_power,player_power,winrate,contributor_id,samples,first_seen,last_seen)
                          VALUES(?,?,?,?,?,?,1,?,?)
                          ON CONFLICT(pit_type,level,enemy_power,contributor_id) DO UPDATE SET
                            player_power=COALESCE(excluded.player_power,pit_power_observations.player_power),
                            winrate=COALESCE(excluded.winrate,pit_power_observations.winrate),
                            samples=pit_power_observations.samples+1,last_seen=excluded.last_seen""",
                       (row["pit_type"], row["level"], row["enemy_power"], row["player_power"],
                        row["winrate"], contributor, now, now))
        total = db.execute("SELECT COUNT(*) FROM pit_power_observations").fetchone()[0]
    return {"ok": True, "accepted": len(observations), "total": total}


def pit_power_catalog() -> list[dict]:
    with db_session() as db:
        rows = db.execute("""SELECT pit_type,level,enemy_power,
                              COUNT(DISTINCT contributor_id) AS players,SUM(samples) AS samples,
                              AVG(winrate) AS observed_winrate,MAX(last_seen) AS last_seen
                           FROM pit_power_observations
                           GROUP BY pit_type,level,enemy_power
                           ORDER BY level,pit_type,players DESC,samples DESC""").fetchall()
    return [dict(row) for row in rows]


def normalize_clan_skill_query(clan_key: object, week_key: object) -> tuple[str, str]:
    clan = str(clan_key or "").strip()
    week = str(week_key or "").strip()
    if not ID_RE.fullmatch(clan):
        raise ValueError("invalid clan key")
    try:
        parsed_week = datetime.strptime(week, "%Y-%m-%d")
    except (TypeError, ValueError) as error:
        raise ValueError("invalid week key") from error
    if parsed_week.weekday() != 0:
        raise ValueError("week key must be Monday")
    return clan, week


def normalize_clan_skill_submission(clan_key: object, week_key: object,
                                    nickname: object, levels: object) -> tuple[str, str, str, dict[str, int]]:
    clan, week = normalize_clan_skill_query(clan_key, week_key)
    name = str(nickname or "").strip()[:100]
    if not name:
        raise ValueError("nickname is required")
    if not isinstance(levels, dict) or len(levels) > 300:
        raise ValueError("invalid skill levels")
    normalized: dict[str, int] = {}
    for raw_id, raw_value in levels.items():
        line_id = str(raw_id or "").strip()
        if not ID_RE.fullmatch(line_id):
            continue
        try:
            value = int(raw_value)
        except (TypeError, ValueError):
            continue
        if 0 <= value <= 100_000:
            normalized[line_id] = value
    if not normalized:
        raise ValueError("skill levels are empty")
    return clan, week, name, normalized


def clan_skill_players(clan_key: str, week_key: str) -> list[dict]:
    clan, week = normalize_clan_skill_query(clan_key, week_key)
    with db_session() as db:
        rows = db.execute("""SELECT nickname,levels_json,scanned_at
                             FROM clan_skill_snapshots
                             WHERE clan_key=? AND week_key=?""", (clan, week)).fetchall()
    players = []
    for row in rows:
        try:
            levels = json.loads(row["levels_json"])
        except (json.JSONDecodeError, TypeError):
            levels = {}
        levels = {str(key): max(0, int(value)) for key, value in levels.items()
                  if isinstance(value, (int, float)) and not isinstance(value, bool)}
        players.append({"nickname": row["nickname"], "levels": levels,
                        "total": sum(levels.values()), "scanned_at": row["scanned_at"]})
    players.sort(key=lambda row: (-row["total"], row["nickname"].casefold()))
    return players


def submit_clan_skill_snapshot(player_id: str, clan_key: object, week_key: object,
                               nickname: object, levels: object) -> dict:
    clan, week, name, normalized = normalize_clan_skill_submission(
        clan_key, week_key, nickname, levels)
    now = utc_now()
    encoded = json.dumps(normalized, ensure_ascii=False, separators=(",", ":"))
    with db_session() as db:
        db.execute("""INSERT INTO clan_skill_snapshots(
                        clan_key,week_key,player_id,nickname,levels_json,scanned_at)
                      VALUES(?,?,?,?,?,?)
                      ON CONFLICT(clan_key,week_key,player_id) DO UPDATE SET
                        nickname=excluded.nickname,levels_json=excluded.levels_json,
                        scanned_at=excluded.scanned_at""",
                   (clan, week, player_id, name, encoded, now))
    players = clan_skill_players(clan, week)
    return {"ok": True, "clan_key": clan, "week_key": week,
            "players": players, "total_players": len(players)}


def normalize_clan_full_snapshot(value: object) -> dict:
    if not isinstance(value, dict):
        raise ValueError("snapshot object required")
    if int(value.get("schema_version") or 0) != 2:
        raise ValueError("unsupported snapshot schema")
    clan, week = normalize_clan_skill_query(value.get("clan_key"), value.get("week_key"))
    try:
        captured_at = int(value.get("captured_at") or utc_now())
        participants_count = int(value.get("participants_count") or 0)
    except (TypeError, ValueError) as error:
        raise ValueError("invalid snapshot metadata") from error
    if captured_at <= 0 or captured_at > utc_now() + 86400:
        raise ValueError("invalid captured_at")
    if not 0 <= participants_count <= 500:
        raise ValueError("invalid participants_count")

    def player_row(raw: object, total_key: str = "value") -> dict | None:
        if not isinstance(raw, dict):
            return None
        nickname = str(raw.get("nickname") or raw.get("name") or "").strip()[:100]
        player_id = str(raw.get("player_id") or raw.get("playerId") or "").strip()[:128]
        if not nickname:
            return None
        try:
            amount = int(raw.get(total_key) or 0)
        except (TypeError, ValueError):
            amount = 0
        amount = max(0, min(amount, 1_000_000))
        result = {"player_id": player_id, "nickname": nickname, total_key: amount}
        if total_key == "value":
            for key in ("applied", "pending"):
                try:
                    result[key] = max(0, min(int(raw.get(key) or 0), 1_000_000))
                except (TypeError, ValueError):
                    result[key] = 0
        return result

    overall = []
    for raw in (value.get("overall") if isinstance(value.get("overall"), list) else [])[:500]:
        row = player_row(raw, "total")
        if row:
            overall.append(row)

    blocks = []
    source_blocks = value.get("blocks") if isinstance(value.get("blocks"), list) else []
    if not source_blocks or len(source_blocks) > 100:
        raise ValueError("invalid clan blocks")
    for block_index, raw_block in enumerate(source_blocks):
        if not isinstance(raw_block, dict):
            continue
        block_id = str(raw_block.get("id") or block_index)[:128]
        title = str(raw_block.get("title") or raw_block.get("name") or f"Block {block_index + 1}").strip()[:200]
        icon = str(raw_block.get("icon") or "").strip()[:1000]
        branches = []
        source_branches = raw_block.get("branches") if isinstance(raw_block.get("branches"), list) else []
        if len(source_branches) > 200:
            raise ValueError("too many clan branches")
        for branch_index, raw_branch in enumerate(source_branches):
            if not isinstance(raw_branch, dict):
                continue
            branch_id = str(raw_branch.get("id") or f"{block_id}-{branch_index}")[:128]
            branch_title = str(raw_branch.get("title") or raw_branch.get("name") or f"Branch {branch_index + 1}").strip()[:200]
            branch_icon = str(raw_branch.get("icon") or "").strip()[:1000]
            try:
                total = max(0, min(int(raw_branch.get("total") or 0), 1_000_000))
                player_level = max(0, min(int(raw_branch.get("player_level") or 0), 1_000_000))
            except (TypeError, ValueError):
                total = player_level = 0
            players = []
            source_players = raw_branch.get("players") if isinstance(raw_branch.get("players"), list) else []
            if len(source_players) > 500:
                raise ValueError("too many branch players")
            for raw_player in source_players:
                row = player_row(raw_player, "value")
                if row:
                    players.append(row)
            branches.append({
                "id": branch_id, "title": branch_title, "icon": branch_icon,
                "total": total, "player_level": player_level,
                "participants_count": len(players), "players": players,
            })
        blocks.append({"id": block_id, "title": title, "icon": icon, "branches": branches})

    normalized = {
        "schema_version": 2, "clan_key": clan, "week_key": week,
        "captured_at": captured_at, "participants_count": participants_count or len(overall),
        "overall": overall, "blocks": blocks,
    }
    encoded = json.dumps(normalized, ensure_ascii=False, separators=(",", ":"))
    if len(encoded.encode("utf-8")) > 1_200_000:
        raise ValueError("snapshot too large")
    return normalized

    player_id = contributor_id(player_id)
    added = 0

    if not ID_RE.fullmatch(area_id) or not ID_RE.fullmatch(city_id):
        raise ValueError("invalid map ids")
    buildings = value.get("buildings")
    if not isinstance(buildings, list) or len(buildings) > 5000:
        raise ValueError("invalid map buildings")
    normalized = []
    for row in buildings:
        if not isinstance(row, dict): continue
        building_id = str(row.get("building_id", ""))
        if not ID_RE.fullmatch(building_id): continue
        rooms = row.get("room_count")
        rooms = None if rooms is None else max(0, min(20, int(rooms)))
        normalized.append({"building_id":building_id, "opened":bool(row.get("opened")), "room_count":rooms,
                           "has_events":bool(row.get("has_events")), "is_invest":bool(row.get("is_invest")),
                           "tier":None if row.get("tier") is None else max(0,min(99,int(row.get("tier")))),
                           "faction":str(row.get("faction", ""))[:40], "building_type":str(row.get("building_type", ""))[:100]})
    return {"area_id":area_id, "city_id":city_id, "city_name":str(value.get("city_name", ""))[:120],
            "x":None if value.get("x") is None else int(value.get("x")), "y":None if value.get("y") is None else int(value.get("y")),
            "invest_count":max(0,int(value.get("invest_count") or 0)), "expected_buildings":max(0,int(value.get("expected_buildings") or len(normalized))),
            "buildings":normalized}


def map_city_key(value: str) -> str:
    return re.sub(r"\s+", " ", str(value or "").strip()).casefold()


def rumor_city_key(value: str) -> str:
    key = re.sub(r"[^a-zа-я0-9]+", " ", str(value or "").casefold().replace("ё", "е")).strip()
    aliases = {
        "москва":"moscow", "санкт петербург":"spb", "saint petersburg":"spb", "st petersburg":"spb",
        "минск":"minsk", "берлин":"berlin", "лондон":"london", "париж":"paris", "дубай":"dubai",
        "токио":"tokyo", "сингапур":"singapore", "нью йорк":"new york", "лос анджелес":"los angeles",
        "бостон":"boston", "вашингтон":"washington", "рио":"rio", "rio de janeiro":"rio",
        "рио де жанейро":"rio", "лагос":"lagos", "сидней":"sydney", "окленд":"auckland",
    }
    return aliases.get(key, key)


def enrich_rumor_routes(db: sqlite3.Connection, routes: list[dict]) -> list[dict]:
    areas = db.execute("SELECT area_id,city_id,city_name,x,y FROM map_areas WHERE x IS NOT NULL AND y IS NOT NULL").fetchall()
    alias_rows = db.execute("SELECT source_area_id,canonical_area_id FROM map_area_aliases").fetchall()
    aliases: dict[str, list[str]] = {}
    for row in alias_rows:
        aliases.setdefault(str(row["canonical_area_id"]), []).append(str(row["source_area_id"]))
    result = json.loads(json.dumps(routes, ensure_ascii=False))
    for route in result:
        wanted_city = rumor_city_key(route.get("city", ""))
        for point in route.get("points", []):
            matches = []
            for area in areas:
                city_keys = (rumor_city_key(area["city_name"]), rumor_city_key(area["city_id"]))
                if int(area["x"]) == int(point.get("x", -1)) and int(area["y"]) == int(point.get("y", -1)) and any(
                    wanted_city and (wanted_city in key or key in wanted_city) for key in city_keys if key
                ):
                    canonical = str(area["area_id"])
                    matches.extend(aliases.get(canonical, []))
                    matches.append(canonical)
            point["area_ids"] = list(dict.fromkeys(matches))
    return result


def canonical_map_area(db: sqlite3.Connection, area: dict) -> str:
    alias = db.execute("SELECT canonical_area_id FROM map_area_aliases WHERE source_area_id=?",
                       (area["area_id"],)).fetchone()
    if alias:
        return str(alias[0])
    if area["x"] is None or area["y"] is None:
        return area["area_id"]
    candidates = db.execute("SELECT area_id,city_id,city_name FROM map_areas WHERE x=? AND y=?",
                            (area["x"], area["y"])).fetchall()
    incoming_name = map_city_key(area["city_name"])
    for row in candidates:
        if str(row["city_id"]) == area["city_id"] or (incoming_name and map_city_key(row["city_name"]) == incoming_name):
            return str(row["area_id"])
    return area["area_id"]


def map_completion(db: sqlite3.Connection, area_id: str) -> tuple[int, int, float]:
    area = db.execute("SELECT expected_buildings FROM map_areas WHERE area_id=?", (area_id,)).fetchone()
    counts = db.execute("""SELECT COUNT(*) total,SUM(CASE WHEN room_count IS NOT NULL THEN 1 ELSE 0 END) known
                           FROM map_buildings WHERE area_id=?""", (area_id,)).fetchone()
    total = max(int((area or {"expected_buildings": 0})["expected_buildings"] or 0), int(counts["total"] or 0))
    known = int(counts["known"] or 0)
    return known, total, round(100 * known / total, 1) if total else 0.0


def submit_map_area(player_id: str, value: object) -> dict:
    area, now = normalize_map_area(value), utc_now()
    player_id = contributor_id(player_id)
    with db_session() as db:
        canonical_id = canonical_map_area(db, area)
        existing = db.execute("SELECT 1 FROM map_areas WHERE area_id=?", (canonical_id,)).fetchone()
        if existing:
            known, total, progress = map_completion(db, canonical_id)
            db.execute("""INSERT INTO map_area_aliases(source_area_id,canonical_area_id,first_seen,last_seen)
                          VALUES(?,?,?,?) ON CONFLICT(source_area_id) DO UPDATE SET
                          canonical_area_id=excluded.canonical_area_id,last_seen=excluded.last_seen""",
                       (area["area_id"], canonical_id, now, now))
            if total and known >= total:
                return {"ok":True,"area_id":canonical_id,"source_area_id":area["area_id"],"ignored_complete":True,
                        "progress":progress,"buildings":total,"added_buildings":0,"enriched_buildings":0}
        db.execute("""INSERT INTO map_areas(area_id,city_id,city_name,x,y,invest_count,expected_buildings,last_player_id,last_seen)
                    VALUES(?,?,?,?,?,?,?,?,?) ON CONFLICT(area_id) DO UPDATE SET
                    city_name=CASE WHEN map_areas.city_name='' AND excluded.city_name<>'' THEN excluded.city_name ELSE map_areas.city_name END,
                    x=COALESCE(map_areas.x,excluded.x),y=COALESCE(map_areas.y,excluded.y),invest_count=MAX(map_areas.invest_count,excluded.invest_count),
                    expected_buildings=MAX(map_areas.expected_buildings,excluded.expected_buildings),last_player_id=excluded.last_player_id,last_seen=excluded.last_seen""",
                   (canonical_id,area["city_id"],area["city_name"],area["x"],area["y"],area["invest_count"],area["expected_buildings"],player_id,now))
        db.execute("""INSERT INTO map_area_aliases(source_area_id,canonical_area_id,first_seen,last_seen)
                      VALUES(?,?,?,?) ON CONFLICT(source_area_id) DO UPDATE SET
                      canonical_area_id=excluded.canonical_area_id,last_seen=excluded.last_seen""",
                   (area["area_id"], canonical_id, now, now))
        db.execute("""INSERT OR IGNORE INTO map_area_aliases(source_area_id,canonical_area_id,first_seen,last_seen)
                      VALUES(?,?,?,?)""", (canonical_id, canonical_id, now, now))
        added = enriched = 0
        for row in area["buildings"]:
            before = db.execute("SELECT opened,room_count,has_events,is_invest,tier,faction,building_type FROM map_buildings WHERE area_id=? AND building_id=?",
                                (canonical_id, row["building_id"])).fetchone()
            if not before:
                added += 1
            elif ((before["room_count"] is None and row["room_count"] is not None) or
                  (not before["opened"] and row["opened"]) or (before["tier"] is None and row["tier"] is not None) or
                  (not before["faction"] and row["faction"]) or (not before["building_type"] and row["building_type"])):
                enriched += 1
            db.execute("""INSERT INTO map_buildings(area_id,building_id,opened,room_count,has_events,is_invest,tier,faction,building_type,last_player_id,last_seen)
                        VALUES(?,?,?,?,?,?,?,?,?,?,?) ON CONFLICT(area_id,building_id) DO UPDATE SET
                        opened=MAX(map_buildings.opened,excluded.opened),room_count=COALESCE(map_buildings.room_count,excluded.room_count),
                        has_events=MAX(map_buildings.has_events,excluded.has_events),is_invest=MAX(map_buildings.is_invest,excluded.is_invest),
                        tier=COALESCE(map_buildings.tier,excluded.tier),faction=CASE WHEN map_buildings.faction='' THEN excluded.faction ELSE map_buildings.faction END,
                        building_type=CASE WHEN map_buildings.building_type='' THEN excluded.building_type ELSE map_buildings.building_type END,
                        last_player_id=CASE WHEN map_buildings.room_count IS NULL AND excluded.room_count IS NOT NULL THEN excluded.last_player_id ELSE map_buildings.last_player_id END,
                        last_seen=MAX(map_buildings.last_seen,excluded.last_seen)""",
                       (canonical_id,row["building_id"],int(row["opened"]),row["room_count"],int(row["has_events"]),int(row["is_invest"]),
                        row["tier"],row["faction"],row["building_type"],player_id,now))
        known, total, progress = map_completion(db, canonical_id)
    return {"ok":True,"area_id":canonical_id,"source_area_id":area["area_id"],"ignored_complete":False,
            "progress":progress,"buildings":total,"known_buildings":known,"added_buildings":added,"enriched_buildings":enriched}

def import_kokkaras_map(value: object) -> dict:
    """Import one authorized HK Maps export without storing login/session data."""
    if not isinstance(value, dict) or not isinstance(value.get("meta"), dict):
        raise ValueError("invalid Kokkaras map export")
    meta = value["meta"]
    collection = value.get("buildingsGeoJSON")
    features = collection.get("features") if isinstance(collection, dict) else None
    if not isinstance(features, list):
        raise ValueError("missing buildings GeoJSON")
    buildings = []
    for feature in features:
        properties = feature.get("properties") if isinstance(feature, dict) else None
        if not isinstance(properties, dict) or not properties.get("is_game") or not properties.get("building_id"):
            continue
        crystals = properties.get("crystals")
        buildings.append({
            "building_id": str(properties["building_id"]),
            "opened": crystals is not None,
            "room_count": crystals,
            "has_events": crystals is not None and int(crystals) > 0,
            "is_invest": bool(properties.get("is_investment")),
            "tier": None,
            "faction": str(properties.get("faction") or ""),
            "building_type": str(properties.get("building_generator") or ""),
        })
    # HK Maps labels coordinates as Y:X, while the mobile UI displays x:y.
    area = {
        "area_id": str(meta.get("area_id") or ""),
        "city_id": str(meta.get("city_id") or f"kokkaras:{meta.get('city_name', '')}"),
        "city_name": str(meta.get("city_name") or ""),
        "x": meta.get("y"),
        "y": meta.get("x"),
        "invest_count": int(meta.get("invest_count") or 0),
        "expected_buildings": int(meta.get("expected_buildings") or len(buildings)),
        "buildings": buildings,
    }
    result = submit_map_area("kokkaras-import", area)
    result.update({"source": "kokkaras", "known": sum(row["room_count"] is not None for row in buildings)})
    return result


def map_index() -> list[dict]:
    with db_session() as db:
        rows = db.execute("""SELECT a.*,COUNT(b.building_id) buildings,SUM(b.opened) opened,
          SUM(CASE WHEN b.building_id IS NOT NULL AND b.room_count IS NULL THEN 1 ELSE 0 END) unexplored,
          SUM(CASE WHEN b.room_count IS NOT NULL THEN 1 ELSE 0 END) known,
          SUM(CASE WHEN b.room_count>0 THEN 1 ELSE 0 END) buildings_with_rooms,COALESCE(SUM(b.room_count),0) total_rooms,
          SUM(CASE WHEN b.room_count=0 THEN 1 ELSE 0 END) room0,SUM(CASE WHEN b.room_count=1 THEN 1 ELSE 0 END) room1,
          SUM(CASE WHEN b.room_count=2 THEN 1 ELSE 0 END) room2,SUM(CASE WHEN b.room_count=3 THEN 1 ELSE 0 END) room3,
          SUM(CASE WHEN b.room_count=4 THEN 1 ELSE 0 END) room4,SUM(CASE WHEN b.room_count>=5 THEN 1 ELSE 0 END) room5
          FROM map_areas a LEFT JOIN map_buildings b ON b.area_id=a.area_id GROUP BY a.area_id""").fetchall()
        aliases = {}
        for row in db.execute("SELECT source_area_id,canonical_area_id FROM map_area_aliases ORDER BY first_seen"):
            aliases.setdefault(str(row["canonical_area_id"]), []).append(str(row["source_area_id"]))
    result=[]
    for row in rows:
        item=dict(row); total=max(item["expected_buildings"],item["buildings"] or 0); known=int(item["known"] or 0)
        item["progress"]=round(100*known/total,1) if total else 0; item["rating"]=(item["total_rooms"] or 0)*2+(item["buildings_with_rooms"] or 0)
        item["aliases"] = aliases.get(str(item["area_id"]), [str(item["area_id"])])
        result.append(item)
    return sorted(result,key=lambda row:(-row["rating"],row["city_name"],row["x"] or 0,row["y"] or 0))

def merge_existing_map_duplicates() -> dict:
    groups: dict[tuple[str, int, int], list[str]] = {}
    merged = removed = 0
    with db_session() as db:
        for row in db.execute("SELECT area_id,city_id,city_name,x,y FROM map_areas WHERE x IS NOT NULL AND y IS NOT NULL"):
            city = map_city_key(row["city_name"]) or map_city_key(row["city_id"])
            groups.setdefault((city, int(row["x"]), int(row["y"])), []).append(str(row["area_id"]))
        for area_ids in groups.values():
            if len(area_ids) < 2:
                continue
            ranked = sorted(area_ids, key=lambda value: map_completion(db, value), reverse=True)
            winner = ranked[0]
            for loser in ranked[1:]:
                for row in db.execute("SELECT * FROM map_buildings WHERE area_id=?", (loser,)).fetchall():
                    db.execute("""INSERT INTO map_buildings(area_id,building_id,opened,room_count,has_events,is_invest,tier,faction,building_type,last_player_id,last_seen)
                                VALUES(?,?,?,?,?,?,?,?,?,?,?) ON CONFLICT(area_id,building_id) DO UPDATE SET
                                opened=MAX(map_buildings.opened,excluded.opened),room_count=COALESCE(map_buildings.room_count,excluded.room_count),
                                has_events=MAX(map_buildings.has_events,excluded.has_events),is_invest=MAX(map_buildings.is_invest,excluded.is_invest),
                                tier=COALESCE(map_buildings.tier,excluded.tier),faction=CASE WHEN map_buildings.faction='' THEN excluded.faction ELSE map_buildings.faction END,
                                building_type=CASE WHEN map_buildings.building_type='' THEN excluded.building_type ELSE map_buildings.building_type END,
                                last_player_id=CASE WHEN map_buildings.room_count IS NULL AND excluded.room_count IS NOT NULL THEN excluded.last_player_id ELSE map_buildings.last_player_id END,
                                last_seen=MAX(map_buildings.last_seen,excluded.last_seen)""",
                               (winner,row["building_id"],row["opened"],row["room_count"],row["has_events"],row["is_invest"],row["tier"],
                                row["faction"],row["building_type"],row["last_player_id"],row["last_seen"]))
                loser_area = db.execute("SELECT * FROM map_areas WHERE area_id=?", (loser,)).fetchone()
                db.execute("""UPDATE map_areas SET invest_count=MAX(invest_count,?),expected_buildings=MAX(expected_buildings,?),
                              last_seen=MAX(last_seen,?) WHERE area_id=?""",
                           (loser_area["invest_count"], loser_area["expected_buildings"], loser_area["last_seen"], winner))
                db.execute("UPDATE map_area_aliases SET canonical_area_id=? WHERE canonical_area_id=?", (winner, loser))
                db.execute("""INSERT INTO map_area_aliases(source_area_id,canonical_area_id,first_seen,last_seen) VALUES(?,?,?,?)
                              ON CONFLICT(source_area_id) DO UPDATE SET canonical_area_id=excluded.canonical_area_id,last_seen=excluded.last_seen""",
                           (loser, winner, utc_now(), utc_now()))
                removed += db.execute("DELETE FROM map_areas WHERE area_id=?", (loser,)).rowcount
                merged += 1
    return {"groups": sum(1 for values in groups.values() if len(values) > 1), "merged": merged, "removed": removed}

RATING_KINDS = frozenset({
    "influence", "power", "clans", "alliances",
    "alliance_power", "alliance_influence", "alliance_defense",
})


def public_snapshot_member(player_id: str) -> bool:

        row = db.execute("""SELECT 1 FROM clan_members
                            WHERE linked_player_id=? AND active=1""", (player_id,)).fetchone()

def clean_public_name(value: object, maximum: int = 100) -> str:
    text = re.sub(r"[\x00-\x1f\x7f]+", " ", str(value or "")).strip()
    return re.sub(r"\s+", " ", text)[:maximum]


def normalize_clan_war(value: object) -> dict | None:
    if value in (None, False, {}):
        return None
    if not isinstance(value, dict):
        raise ValueError("invalid clan war")
    result = {
        "our_clan": clean_public_name(value.get("our_clan")),
        "opponent": clean_public_name(value.get("opponent")),
        "status": clean_public_name(value.get("status"), 32) or "active",
    }
    for key in ("our_score", "opponent_score", "started_at", "ends_at"):
        try:
            number = int(value.get(key) or 0)
        except (TypeError, ValueError):
            number = 0
        result[key] = max(0, min(number, 10**18))
    if not result["our_clan"] or not result["opponent"]:
        raise ValueError("clan names required")
    return result


def normalize_rating_rows(kind: str, value: object) -> list[dict]:
    if kind not in RATING_KINDS or not isinstance(value, list) or len(value) > 100:
        raise ValueError("invalid ratings")
    rows, used = [], set()
    for raw in value:
        if not isinstance(raw, dict):
            continue
        try:
            rank = int(raw.get("rank"))
            score = float(raw.get("value"))
        except (TypeError, ValueError):
            continue
        name = clean_public_name(raw.get("name"))
        if not 1 <= rank <= 100 or rank in used or not name or not 0 <= score <= 10**30:
            continue
        used.add(rank)
        rows.append({"rank": rank, "name": name, "value": int(score) if score.is_integer() else score})
    return sorted(rows, key=lambda row: row["rank"])


def store_public_snapshot(player_id: str, document: object) -> dict:
    if not public_snapshot_member(player_id):
        raise PermissionError("member not allowed")
    if not isinstance(document, dict):
        raise ValueError("snapshot object required")
    if any(key.lower() in {"token", "authorization", "jwt", "participants", "members", "damage"}
           for key in document):
        raise ValueError("sensitive snapshot field")
    now, saved_war, saved_ratings = utc_now(), False, []
    with db_session() as db:
        if "war" in document:
            war = normalize_clan_war(document.get("war"))
            db.execute("DELETE FROM public_clan_war_snapshot")
            if war:
                db.execute("""INSERT INTO public_clan_war_snapshot
                              (singleton,snapshot_json,source_player_id,updated_at) VALUES(1,?,?,?)""",
                           (json.dumps(war, ensure_ascii=False, separators=(",", ":")),
                            contributor_id(player_id), now))
            saved_war = True
        ratings = document.get("ratings")
        if ratings is not None:
            if not isinstance(ratings, dict):
                raise ValueError("ratings object required")
            for kind, raw_rows in ratings.items():
                if kind not in RATING_KINDS:
                    continue
                rows = normalize_rating_rows(kind, raw_rows)
                db.execute("DELETE FROM public_rating_snapshots WHERE kind=?", (kind,))
                db.executemany("""INSERT INTO public_rating_snapshots
                                  (kind,rank,name,value,source_player_id,updated_at)
                                  VALUES(?,?,?,?,?,?)""",
                               [(kind, row["rank"], row["name"], row["value"], contributor_id(player_id), now)
                                for row in rows])
                saved_ratings.append(kind)
    if not saved_war and not saved_ratings:
        raise ValueError("empty snapshot")
    return {"ok": True, "war": saved_war, "ratings": saved_ratings, "updated_at": now}


def public_clan_war() -> dict:
    with db_session() as db:
        row = db.execute("SELECT snapshot_json,updated_at FROM public_clan_war_snapshot WHERE singleton=1").fetchone()
    if not row:
        return {"ok": True, "active": False, "war": None, "updated_at": 0, "stale": False}
    try:
        war = json.loads(row["snapshot_json"])
    except json.JSONDecodeError:
        war = None
    stale = row["updated_at"] < utc_now() - 1800
    if war:
        war = {**war, "active": True, "updated_at": row["updated_at"], "stale": stale}
    return {"ok": True, "active": bool(war), "war": war, "updated_at": row["updated_at"], "stale": stale}


def public_ratings(kind: str) -> dict:
    if kind not in RATING_KINDS:
        raise ValueError("invalid rating kind")
    with db_session() as db:
        rows = db.execute("""SELECT rank,name,value,updated_at FROM public_rating_snapshots
                             WHERE kind=? ORDER BY rank LIMIT 100""", (kind,)).fetchall()
    updated_at = max((row["updated_at"] for row in rows), default=0)
    return {"ok": True, "kind": kind, "updated_at": updated_at,
            "stale": bool(updated_at and updated_at < utc_now() - 1800),
            "rows": [{"rank": row["rank"], "name": row["name"], "value": row["value"]} for row in rows]}
