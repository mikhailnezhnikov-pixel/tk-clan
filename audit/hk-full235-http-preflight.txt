def cabinet_maps(member: dict) -> dict:
    ensure_scoped_hk_map_access_schema()
    with db_session() as db:
        rows = db.execute("SELECT * FROM hk_maps_catalog ORDER BY score DESC,city,grid").fetchall()
        if not member.get('maps_manage'):
            telegram_id = str(member.get('telegram_id') or '')
            cities, map_rules = _hk_member_rules(db, telegram_id)
            rows = [row for row in rows if map_rules.get(row['map_key'], row['city'] in cities)]
            if not rows:
                raise PermissionError('maps access denied')
        members = []
        if member.get('maps_manage'):
            raw_members = [dict(row) for row in db.execute(
                """SELECT telegram_id,note,first_name,username,linked_player_id,active,maps_access,maps_manage
                   FROM clan_members WHERE active=1 ORDER BY first_name,username,note,telegram_id""").fetchall()]
            rule_rows = db.execute("SELECT telegram_id,scope_type,scope_value,allowed FROM hk_map_access_rules").fetchall()
            by_member: dict[str, dict] = {}
            for rule in rule_rows:
                bucket = by_member.setdefault(rule['telegram_id'], {'cities': [], 'maps': {}})
                if rule['scope_type'] == 'city' and rule['allowed']:
                    bucket['cities'].append(rule['scope_value'])
                elif rule['scope_type'] == 'map':
                    bucket['maps'][rule['scope_value']] = bool(rule['allowed'])
            for item in raw_members:
                rules = by_member.get(item['telegram_id'], {'cities': [], 'maps': {}})
                rules['cities'] = sorted(set(rules['cities']))
                item['map_rules'] = rules
                members.append(item)
    return {'ok': True, 'can_manage': bool(member.get('maps_manage')),
            'maps': [hk_map_row(row) for row in rows], 'members': members}

### cabinet_maps
def cabinet_maps(member: dict) -> dict:
    """One Personal Cabinet list: website geometry + canonical/script knowledge."""
    try:
        website = _cabinet_maps_website_only(member)
    except PermissionError:
        website = {"ok": True, "can_manage": bool(member.get("maps_manage")), "maps": [], "members": []}

    website_rows = [dict(row) for row in website.get("maps") or []]
    canonical_rows = map_index()
    canonical_by_id = {str(row.get("area_id") or ""): row for row in canonical_rows}

    with db_session() as db:
        link_rows = db.execute(
            "SELECT map_key,canonical_area_id FROM hk_map_area_links"
        ).fetchall()
        links = {str(row["map_key"]): str(row["canonical_area_id"]) for row in link_rows}
        cities = set()
        map_rules = {}
        if not member.get("maps_manage"):
            cities, map_rules = _hk_member_rules(db, str(member.get("telegram_id") or ""))

    grouped: dict[str, list[dict]] = {}
    unlinked: list[dict] = []
    for row in website_rows:
        key = str(row.get("key") or "")
        area_id = links.get(key, "")
        if area_id and area_id in canonical_by_id:
            grouped.setdefault(area_id, []).append(row)
        else:
            unlinked.append({
                **row,
                "website_key": key,
                "website_keys": [key] if key else [],
                "area_id": "",
                "canonical_area_id": "",
                "source_kind": "website",
                "source_label": "Сайт",
                "canonical_grid": "",
                "shared_known": 0,
                "aliases": [],
            })

    result = []
    used_canonical = set()
    for area_id, rows in grouped.items():
        # If several website visualizations ever point at one canonical district,
        # keep one row and choose the richest website visualization as primary.
        primary = max(
            rows,
            key=lambda row: (
                int(bool(row.get("open_url"))),
                float(row.get("progress") or 0),
                int(row.get("score") or 0),
            ),
        )
        keys = sorted({str(row.get("key") or "") for row in rows if row.get("key")})
        result.append(_cabinet_unified_canonical_row(canonical_by_id[area_id], primary, keys))
        used_canonical.add(area_id)

    result.extend(unlinked)

    for row in canonical_rows:
        area_id = str(row.get("area_id") or "")
        if not area_id or area_id in used_canonical:
            continue
        city = str(row.get("city_name") or row.get("city_id") or "")
        synthetic_key = f"canonical:{area_id}"
        if not member.get("maps_manage"):
            allowed = bool(map_rules.get(synthetic_key, city in cities))
            if not allowed:
                continue
        result.append(_cabinet_unified_canonical_row(row))

    if not result:
        raise PermissionError("maps access denied")

    source_counts = {"website":0,"script":0,"combined":0}
    for row in result:
        kind = str(row.get("source_kind") or "website")
        source_counts[kind] = source_counts.get(kind,0) + 1

    result.sort(key=lambda row: (
        -int(row.get("shared_known") or 0),
        -float(row.get("progress") or 0),
        str(row.get("city") or ""),
        str(row.get("grid") or ""),
    ))
    return {
        "ok": True,
        "can_manage": bool(member.get("maps_manage")),
        "maps": result,
        "members": website.get("members") or [],
        "unified": True,
        "source_counts": source_counts,
        "website_maps": len(website_rows),
        "canonical_maps": len(canonical_rows),
    }

class Handler(BaseHTTPRequestHandler):
    server_version = "HKLicense/1.0"

    def log_message(self, fmt: str, *args) -> None:
        safe_path = urlsplit(self.path).path
        print(f"{self.client_ip()} - {self.command} {safe_path} - {args[1] if len(args) > 1 else '-'}", flush=True)

    def client_ip(self) -> str:
        peer = self.client_address[0]
        candidate = peer
        try:
            if ipaddress.ip_address(peer).is_loopback:
                candidate = (self.headers.get("X-Real-IP") or self.headers.get("X-Forwarded-For", "").split(",")[0] or peer).strip()
            return str(ipaddress.ip_address(candidate))
        except ValueError:
            return peer

    def send_common(self, status: int, content_type: str, body: bytes, extra: dict | None = None) -> None:
        extra_headers = dict(extra or {})
        content_security_policy = extra_headers.pop(
            "Content-Security-Policy",
            "default-src 'self'; style-src 'unsafe-inline'; form-action 'self'; frame-ancestors 'none'",
        )
        cache_control = extra_headers.pop("Cache-Control", "no-store")
        self.send_response(status)
        self.send_header("Content-Type", content_type)
        self.send_header("Content-Length", str(len(body)))
        self.send_header("Cache-Control", cache_control)
        self.send_header("X-Content-Type-Options", "nosniff")
        self.send_header("X-Frame-Options", "DENY")
        self.send_header("Content-Security-Policy", content_security_policy)
        for name, value in extra_headers.items():
            self.send_header(name, value)
        self.end_headers()
        self.wfile.write(body)

    def send_json(self, status: int, document: dict) -> None:
        origin = self.headers.get("Origin", "")
        extra = {}
        if origin in ALLOWED_ORIGINS:
            extra = {"Access-Control-Allow-Origin": origin, "Vary": "Origin"}
        self.send_common(status, "application/json; charset=utf-8",
                         json.dumps(document, ensure_ascii=False, separators=(",", ":")).encode(), extra)

    def read_json(self, maximum: int = 4096) -> dict:
        length = int(self.headers.get("Content-Length", "0"))
        if length < 1 or length > maximum:
            raise ValueError("invalid body length")
        value = json.loads(self.rfile.read(length))
        if not isinstance(value, dict):
            raise ValueError("object required")
        return value

    def read_form(self) -> dict[str, str]:
        length = min(int(self.headers.get("Content-Length", "0")), 8192)
        values = parse_qs(self.rfile.read(length).decode("utf-8", "replace"), keep_blank_values=True)
        return {key: items[-1] for key, items in values.items()}

    def admin_ok(self) -> bool:
        cookie = SimpleCookie(self.headers.get("Cookie", ""))
        return verify_token(cookie.get("hk_admin").value, "admin") if cookie.get("hk_admin") else False

    def recipe_player(self) -> str | None:
        authorization = self.headers.get("Authorization", "")
        if not authorization.startswith("Bearer "):
            return None
        document = token_document(authorization[7:].strip())
        subject = str((document or {}).get("sub", ""))
        if ":" not in subject:
            return None
        player_id, _device_id = subject.rsplit(":", 1)
        return player_id if ID_RE.fullmatch(player_id) else None

        if not authorization.startswith("Bearer "):

    def admin_action(self, path: str, form: dict[str, str]) -> None:
        if path == "/admin/rumors/publish":
            try:
                route_date, routes = parse_rumor_route(form.get("route_text", ""))
                now = utc_now()
                with db_session() as db:
                    db.execute("""INSERT INTO rumor_routes(route_date,source_text,routes_json,published_at,updated_at)
                                  VALUES(?,?,?,?,?) ON CONFLICT(route_date) DO UPDATE SET source_text=excluded.source_text,
                                  routes_json=excluded.routes_json,published_at=excluded.published_at,updated_at=excluded.updated_at""",
                               (route_date, form.get("route_text", "")[:20000], json.dumps(routes, ensure_ascii=False, separators=(",", ":")), now, now))
                audit("rumors_publish", route_date, self.client_ip(), f"cities={len(routes)} points={sum(len(row['points']) for row in routes)}")
            except ValueError as error:
                audit("rumors_publish_failed", "", self.client_ip(), str(error))
            self.redirect("/admin")
            return
        if path.startswith("/admin/telegram/"):
            telegram_id = form.get("telegram_id", "").strip()
            if not TELEGRAM_ID_RE.fullmatch(telegram_id):
                self.redirect("/admin")
                return
            now = utc_now()
            with db_session() as db:
                if path == "/admin/telegram/add":
                    db.execute("""INSERT INTO clan_members(telegram_id,note,active,created_at,updated_at)
                                  VALUES(?,?,1,?,?) ON CONFLICT(telegram_id) DO UPDATE SET
                                  note=excluded.note,active=1,updated_at=excluded.updated_at""",
                               (telegram_id, form.get("note", "")[:200], now, now))
                elif path == "/admin/telegram/maps-access":
                    row = db.execute("SELECT maps_access FROM clan_members WHERE telegram_id=?", (telegram_id,)).fetchone()
                    if row:
                        db.execute("UPDATE clan_members SET maps_access=?,updated_at=? WHERE telegram_id=?",
                                   (0 if row["maps_access"] else 1, now, telegram_id))
                elif path == "/admin/telegram/maps-manage":
                    row = db.execute("SELECT maps_manage FROM clan_members WHERE telegram_id=?", (telegram_id,)).fetchone()
                    if row:
                        manage = 0 if row["maps_manage"] else 1
                        db.execute("UPDATE clan_members SET maps_manage=?,maps_access=CASE WHEN ?=1 THEN 1 ELSE maps_access END,updated_at=? WHERE telegram_id=?",
                                   (manage, manage, now, telegram_id))
                elif path == "/admin/telegram/toggle":
                    db.execute("UPDATE clan_members SET active=1-active,updated_at=? WHERE telegram_id=?", (now, telegram_id))
                elif path == "/admin/telegram/delete":
                    db.execute("DELETE FROM clan_members WHERE telegram_id=?", (telegram_id,))
            audit(path.removeprefix("/admin/").replace("/", "_"), telegram_id, self.client_ip())
            self.redirect("/admin")
            return
        player_id = form.get("player_id", "").strip()
        if not ID_RE.fullmatch(player_id):
            self.redirect("/admin")
            return
        now = utc_now()
        with db_session() as db:
            if path == "/admin/add":
                expires_text = form.get("expires", "").strip()
                expires_at = int(datetime.strptime(expires_text, "%Y-%m-%d").replace(tzinfo=timezone.utc).timestamp()) if expires_text else None
                db.execute("""INSERT INTO licenses(player_id,note,active,expires_at,created_at,updated_at)
                            VALUES(?,?,1,?,?,?) ON CONFLICT(player_id) DO UPDATE SET note=excluded.note,
                            active=1,expires_at=excluded.expires_at,updated_at=excluded.updated_at""",
                           (player_id, form.get("note", "")[:200], expires_at, now, now))
            elif path == "/admin/toggle":
                db.execute("UPDATE licenses SET active=1-active,updated_at=? WHERE player_id=?", (now, player_id))
            elif path == "/admin/delete":
                db.execute("DELETE FROM licenses WHERE player_id=?", (player_id,))
        audit(path.removeprefix("/admin/").replace("/", "_"), player_id, self.client_ip())
        self.redirect("/admin")

### send_json
def send_json(self, status: int, document: dict) -> None:
        origin = self.headers.get("Origin", "")
        extra = {}
        if origin in ALLOWED_ORIGINS:
            extra = {"Access-Control-Allow-Origin": origin, "Vary": "Origin"}
        self.send_common(status, "application/json; charset=utf-8",
                         json.dumps(document, ensure_ascii=False, separators=(",", ":")).encode(), extra)
